Kit copia e cola — NP2 Calculadora NAF

Este kit é para aplicar manualmente no fork. Não altera seu GitHub. Crie uma branch por bloco e cole o conteúdo nos caminhos indicados.

Premissas importantes

  • O Fator R usa a folha/pró-labore de 12 meses dividida pela receita dos mesmos 12 meses. O formulário recebe valores mensais e anualiza por 12; é uma estimativa que pressupõe valores estáveis.
  • O IRPF progressivo abaixo usa os limites mensais oficiais de 2026, incluindo a redução do imposto até R$ 7.350.
  • O cálculo do INSS mantém a premissa simplificada já usada no projeto (11% sobre o maior entre 28% da renda e salário mínimo R$ 1.621). Confirme essa premissa com o professor. Não use o resultado como apuração real.
  • Em Sobre, substitua EMAIL_NAF_CONFIRMADO por contato confirmado pelo NAF. O mailto: abre o app de e-mail do usuário; envio automático exige serviço e credenciais no backend.

1. Backend: motor de cálculo

Crie backend/src/services/calculoTributario.js:

const sujeitasFatorR = new Set(['psicologo','arquiteto','medico','dentista']);
const tabelas = {
 III: [[180000,.06,0],[360000,.112,9360],[720000,.135,17640],[1800000,.16,35640],[3600000,.21,125640],[4800000,.33,648000]],
 IV: [[180000,.045,0],[360000,.09,8100],[720000,.102,12420],[1800000,.14,39780],[3600000,.22,183780],[4800000,.33,828000]],
 V: [[180000,.155,0],[360000,.18,4500],[720000,.195,9900],[1800000,.205,17100],[3600000,.23,62100],[4800000,.305,540000]],
};
function calcularPF(renda,custos) {
 const base=Math.max(0,renda-custos);
 const faixas=[[2428.80,0,0,'Isento'],[2826.65,.075,182.16,'7,5%'],[3751.05,.15,394.16,'15%'],[4664.68,.225,675.49,'22,5%'],[Infinity,.275,908.73,'27,5%']];
 const [_,aliquota,deducao,nome]=faixas.find(f=>base<=f[0]);
 const antes=Math.max(0,base*aliquota-deducao);
 let reducao=base<=5000?antes:base<=7350?Math.max(0,978.62-.133145*base):0;
 return {base,aliquota,deducao,faixa:nome,antes,reducao2026:reducao,imposto:Math.max(0,antes-reducao)};
}
function faixaAnexo(anexo,receita12) {
 const tabela=tabelas[anexo], faixa=tabela.find(f=>receita12<=f[0])||tabela.at(-1);
 const efetiva=(receita12*faixa[1]-faixa[2])/receita12;
 return {nominal:faixa[1],deducao:faixa[2],efetiva};
}
function calcularComparativo({tipoProfissao,faturamento,custos=0,folhaMensal=0}) {
 const profissao=String(tipoProfissao||'').toLowerCase();
 const renda=Number(faturamento), despesas=Number(custos||0), folha=Number(folhaMensal||0);
 if(!['psicologo','arquiteto','advogado','medico','dentista'].includes(profissao)) throw Error('Selecione uma profissão válida.');
 if(![renda,despesas,folha].every(Number.isFinite)||renda<=0||renda>15000||despesas<0||despesas>renda||folha<0) throw Error('Renda deve ser até R$ 15.000; custos e folha não podem ser negativos.');
 if(sujeitasFatorR.has(profissao)&&folha<=0) throw Error('Informe a folha/pró-labore para calcular o Fator R.');
 const pf=calcularPF(renda,despesas), receita12=renda*12, folha12=folha*12;
 const fatorR=sujeitasFatorR.has(profissao)?folha12/receita12:null;
 const anexo=profissao==='advogado'?'IV':fatorR===null||fatorR>=.28?'III':'V';
 const faixa=faixaAnexo(anexo,receita12), simples=renda*faixa.efetiva;
 const salarioBase=Math.max(renda*.28,1621), inss=salarioBase*.11, totalPJ=simples+inss;
 return {profissao,faturamento:renda,custos:despesas,folhaMensal:folha,receita12,folha12,fatorR,anexo,
   pf,pj:{baseSimples:renda,aliquotaEfetiva:faixa.efetiva,aliquotaNominal:faixa.nominal,deducaoAnexo:faixa.deducao,simples,salarioBase,inss,total:totalPJ},
   impostoPF:pf.imposto,impostoPJ:totalPJ,melhorOpcao:totalPJ<pf.imposto?'Pessoa Jurídica (PJ)':'Pessoa Física (PF)',economiaMensal:Math.abs(totalPJ-pf.imposto)};
}
module.exports={calcularComparativo};

1. Gráfico de ponto de equilíbrio

Este gráfico opcional estima a renda em que a recomendação PF/PJ muda. Para evitar dezenas de chamadas da interface ao servidor, o backend calcula a série inteira em uma única solicitação.

No final de backend/src/services/calculoTributario.js, acrescente:

function gerarPontosEquilibrio({ tipoProfissao, custos = 0, folhaMensal = 0 }) {
  const pontos = [];
  for (let faturamento = 500; faturamento <= 15000; faturamento += 250) {
    const resultado = calcularComparativo({
      tipoProfissao,
      faturamento,
      custos: Math.min(Number(custos || 0), faturamento),
      folhaMensal: Number(folhaMensal || 0),
    });
    pontos.push({
      faturamento,
      impostoPF: resultado.impostoPF,
      impostoPJ: resultado.impostoPJ,
      melhorOpcao: resultado.melhorOpcao,
    });
  }
  return pontos;
}

module.exports.gerarPontosEquilibrio = gerarPontosEquilibrio;

import { useEffect, useState } from 'react';
import api from '../services/api';

export default function PontoEquilibrio({ resultado }) {
  const [pontos, setPontos] = useState([]);
  const [erro, setErro] = useState('');

  useEffect(() => {
    let ativo = true;
    api.post('/calculos/equilibrio', {
      tipoProfissao: resultado.profissao,
      custos: resultado.custos,
      folhaMensal: resultado.folhaMensal || 0,
    }).then(({ data }) => {
      if (ativo) setPontos(data.pontos);
    }).catch(() => {
      if (ativo) setErro('Não foi possível gerar o gráfico.');
    });
    return () => { ativo = false; };
  }, [resultado]);

  if (erro) return <p role="alert">{erro}</p>;
  if (!pontos.length) return <p>Calculando faixa de equilíbrio…</p>;

  const max = Math.max(...pontos.flatMap(p => [p.impostoPF, p.impostoPJ]), 1);
  const w = 720, h = 260, pad = 38;
  const xy = (valor, indice) => [
    pad + indice * (w - 2 * pad) / (pontos.length - 1),
    h - pad - (valor / max) * (h - 2 * pad),
  ];
  const path = (campo) => pontos.map((p, i) => {
    const [x, y] = xy(p[campo], i);
    return (i ? 'L' : 'M') + x + ' ' + y;
  }).join(' ');

  return <section className="my-6 rounded-xl border bg-white p-4">
    <h2 className="mb-2 text-lg font-bold">Ponto de equilíbrio estimado</h2>
    <svg viewBox={'0 0 ' + w + ' ' + h} role="img"
      aria-label="Gráfico de impostos PF e PJ por faturamento mensal">
      <path d={path('impostoPF')} fill="none" stroke="#2563eb" strokeWidth="3" />
      <path d={path('impostoPJ')} fill="none" stroke="#c2410c" strokeWidth="3" />
      <text x={pad} y={h - 8}>R$ 500/mês</text>
      <text x={w - 130} y={h - 8}>R$ 15.000/mês</text>
    </svg>
    <p className="text-sm">Azul: PF. Laranja: PJ. Valores são estimativas mensais.</p>
  </section>;
}

3. Chatbot com IA para dúvidas sobre o comparativo

O chatbot exige uma chave de API e pode gerar cobrança. A chave fica somente no backend, em variável de ambiente; nunca a coloque no React nem a envie para o GitHub. O exemplo usa a Responses API da OpenAI; configure OPENAI_MODEL com um modelo disponível na sua conta.

No backend, crie backend/src/routes/chatRoutes.js:

const express = require('express');
const router = express.Router();

router.post('/', async (req, res) => {
  const pergunta = String(req.body.pergunta || '').trim();
  if (!pergunta || pergunta.length > 1000) {
    return res.status(400).json({ error: 'Escreva uma pergunta de até 1000 caracteres.' });
  }
  if (!process.env.OPENAI_API_KEY || !process.env.OPENAI_MODEL) {
    return res.status(503).json({ error: 'Chatbot ainda não configurado.' });
  }
  try {
    const resposta = await fetch('https://api.openai.com/v1/responses', {
      method: 'POST',
      headers: {
        Authorization: 'Bearer ' + process.env.OPENAI_API_KEY,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        model: process.env.OPENAI_MODEL,
        instructions: 'Responda em português, de forma simples, somente sobre conceitos gerais do comparativo PF/PJ e do resultado informado pelo usuário. Não invente alíquotas nem faça recomendação tributária profissional. Explique que a calculadora é uma estimativa acadêmica e que a pessoa deve consultar um contador.',
        input: pergunta,
        max_output_tokens: 300,
      }),
    });
    const data = await resposta.json();
    if (!resposta.ok) {
      return res.status(502).json({ error: 'O serviço de IA não respondeu. Tente novamente.' });
    }
    const texto = (data.output || [])
      .flatMap(item => item.content || [])
      .filter(item => item.type === 'output_text')
      .map(item => item.text)
      .join('\n');
    return res.json({ resposta: texto || 'Não consegui formular uma resposta. Tente perguntar de outro jeito.' });
  } catch {
    return res.status(502).json({ error: 'Não foi possível acessar o chatbot.' });
  }
});

module.exports = router;

import { useState } from 'react';
import api from '../services/api';

export default function Chatbot() {
  const [pergunta, setPergunta] = useState('');
  const [mensagens, setMensagens] = useState([]);
  const [carregando, setCarregando] = useState(false);

  async function enviar(e) {
    e.preventDefault();
    const texto = pergunta.trim();
    if (!texto || carregando) return;
    setMensagens(atual => [...atual, { autor: 'Você', texto }]);
    setPergunta('');
    setCarregando(true);
    try {
      const { data } = await api.post('/chat', { pergunta: texto });
      setMensagens(atual => [...atual, { autor: 'Assistente', texto: data.resposta }]);
    } catch (error) {
      setMensagens(atual => [...atual, {
        autor: 'Aviso',
        texto: error.response?.data?.error || 'Não foi possível responder agora.',
      }]);
    } finally {
      setCarregando(false);
    }
  }

  return <section className="my-6 rounded-xl border bg-white p-4">
    <h2 className="text-lg font-bold">Tire dúvidas sobre o resultado</h2>
    <div aria-live="polite" className="my-3 max-h-64 space-y-2 overflow-y-auto">
      {mensagens.map((m, i) => <p key={i}><strong>{m.autor}:</strong> {m.texto}</p>)}
      {carregando && <p>Preparando resposta…</p>}
    </div>
    <form onSubmit={enviar} className="flex flex-col gap-2 sm:flex-row">
      <label className="sr-only" htmlFor="pergunta-chat">Sua pergunta</label>
      <input id="pergunta-chat" maxLength="1000" value={pergunta}
        onChange={e => setPergunta(e.target.value)} className="min-w-0 flex-1 rounded border p-3"
        placeholder="Ex.: por que o Fator R muda o anexo?" />
      <button disabled={carregando} className="rounded bg-naf-blue px-4 py-3 font-bold text-white">
        Enviar pergunta
      </button>
    </form>
    <p className="mt-2 text-xs">Respostas automáticas para fins educativos; confirme regras tributárias com contador.</p>
  </section>;
}
